function(shader NAME ENTRY PROFILE)
  set(EXTRA_SHADER_DEPS)
  if(NAME STREQUAL "lattice" OR NAME STREQUAL "plastic_metal")
    list(APPEND EXTRA_SHADER_DEPS "shaders/sculpture_common.hlsli")
  endif()
  if(NAME STREQUAL "ripples")
    list(APPEND EXTRA_SHADER_DEPS "shaders/ripples_surface.hlsli" "shaders/ripples_bottom.hlsli")
  endif()
  if(NAME MATCHES "^tropical_island")
    list(APPEND EXTRA_SHADER_DEPS "shaders/tropical_island_masks.hlsli" "shaders/tropical_island_ocean.hlsli")
  endif()
  if(NAME STREQUAL "tropical_island_composite")
    list(APPEND EXTRA_SHADER_DEPS "shaders/tropical_island.hlsl")
  endif()
  if(NAME STREQUAL "forest_water")
    list(APPEND EXTRA_SHADER_DEPS "shaders/ripples_surface.hlsli")
  endif()
  if(NAME STREQUAL "bloom_mesh" OR NAME STREQUAL "bloom_fold")
    list(APPEND EXTRA_SHADER_DEPS "shaders/bloom_surface.hlsli")
  endif()
  if(NAME MATCHES "^smoke_filament_")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_filament_shared.hlsli" "shaders/smoke_lighting.hlsli")
  endif()
  if(NAME STREQUAL "fluid_multigrid")
    list(APPEND EXTRA_SHADER_DEPS "shaders/fluid.hlsl" "shaders/smoke_transport.hlsli" "shaders/smoke_lighting.hlsli" "shaders/smoke_sources.hlsli" "shaders/smoke_half.hlsli")
  endif()
  if(NAME STREQUAL "aurora")
    list(APPEND EXTRA_SHADER_DEPS "shaders/aurora_volume.hlsli")
  endif()
  if(NAME MATCHES "^(earth|mars|moon|jupiter|mercury|venus|uranus|neptune|pluto)_space$")
    list(APPEND EXTRA_SHADER_DEPS "shaders/planet_sky.hlsli")
  endif()
  if(NAME STREQUAL "saturn")
    list(APPEND EXTRA_SHADER_DEPS "shaders/saturn_common.hlsli" "shaders/orbit_rings.hlsli" "shaders/planet_sky.hlsli")
  endif()
  if(NAME STREQUAL "jupiter" OR NAME STREQUAL "jupiter_space")
    list(APPEND EXTRA_SHADER_DEPS "shaders/jupiter_common.hlsli")
  endif()
  if(NAME STREQUAL "moon" OR NAME STREQUAL "moon_space")
    list(APPEND EXTRA_SHADER_DEPS "shaders/moon_common.hlsli")
  endif()
  if(NAME STREQUAL "earth" OR NAME STREQUAL "earth_space")
    list(APPEND EXTRA_SHADER_DEPS "shaders/earth_common.hlsli")
  endif()
  if(NAME MATCHES "^uranus")
    list(APPEND EXTRA_SHADER_DEPS "shaders/uranus_common.hlsli")
  endif()
  if(NAME MATCHES "^neptune")
    list(APPEND EXTRA_SHADER_DEPS "shaders/neptune_common.hlsli")
  endif()
  if(NAME MATCHES "^venus")
    list(APPEND EXTRA_SHADER_DEPS "shaders/venus_common.hlsli")
  endif()
  if(NAME MATCHES "^pluto")
    list(APPEND EXTRA_SHADER_DEPS "shaders/pluto_common.hlsli")
  endif()
  if(NAME MATCHES "^mercury")
    list(APPEND EXTRA_SHADER_DEPS "shaders/mercury_common.hlsli")
  endif()
  if(NAME STREQUAL "mars" OR NAME STREQUAL "mars_space")
    list(APPEND EXTRA_SHADER_DEPS "shaders/mars_common.hlsli")
  endif()
  if(NAME STREQUAL "planet")
    list(APPEND EXTRA_SHADER_DEPS "shaders/orbit_rings.hlsli")
  endif()
  if(NAME STREQUAL "planet" OR NAME STREQUAL "orbit")
    list(APPEND EXTRA_SHADER_DEPS "shaders/orbit_terrain.hlsli")
  endif()
  if(NAME STREQUAL "smoke" OR NAME STREQUAL "fluid")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_lighting.hlsli")
  endif()
  if(NAME STREQUAL "smoke_detail" OR NAME STREQUAL "smoke_coordinates")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_detail_clock.hlsli")
  endif()
  if(NAME STREQUAL "fluid" OR NAME STREQUAL "smoke_coordinates")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_transport.hlsli")
  endif()
  if(NAME STREQUAL "smoke_atlas_advect")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_atlas.hlsli")
  endif()
  if(NAME STREQUAL "smoke")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_atlas.hlsli")
  endif()
  if(NAME STREQUAL "fluid" OR NAME STREQUAL "smoke_atlas_advect")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_half.hlsli" "shaders/smoke_sources.hlsli")
  endif()
  if(NAME STREQUAL "smoke_atlas_cached" OR NAME STREQUAL "smoke_atlas_cached_curl")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_atlas_advect.hlsl" "shaders/smoke_atlas.hlsli" "shaders/smoke_half.hlsli" "shaders/smoke_sources.hlsli")
  endif()
  if(NAME STREQUAL "smoke_atlas_light")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_atlas.hlsli" "shaders/smoke_lighting.hlsli")
  endif()
  add_custom_command(OUTPUT "${SHADER_DIR}/${NAME}.cso"
    COMMAND "${DXC}" -T ${PROFILE} -E ${ENTRY} -O3 -Ges -Fo "${SHADER_DIR}/${NAME}.cso" "${CMAKE_SOURCE_DIR}/shaders/${NAME}.hlsl"
    DEPENDS "shaders/${NAME}.hlsl" "shaders/common.hlsli" ${EXTRA_SHADER_DEPS} VERBATIM)
  set(SHADER_OUTPUTS ${SHADER_OUTPUTS} "${SHADER_DIR}/${NAME}.cso" PARENT_SCOPE)
endfunction()

# Refresh runtime bytecode even when only HLSL changed and the executable did
# not relink (the executable's POST_BUILD copy alone cannot cover that case).
add_custom_target(worlds_shaders ALL
  COMMAND ${CMAKE_COMMAND} -E copy_directory_if_different "${SHADER_DIR}" "${CMAKE_BINARY_DIR}/$<CONFIG>/shaders"
  DEPENDS ${SHADER_OUTPUTS})
add_library(worlds_core STATIC src/settings.cpp src/output.cpp src/package.cpp src/presets.cpp src/policy.cpp)
target_include_directories(worlds_core PUBLIC src "${DEPS}/json")
target_compile_definitions(worlds_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(worlds_core PUBLIC dxgi shell32 ole32 bcrypt windowscodecs)
add_executable(VolturaWorlds WIN32 src/updates.cpp src/main.cpp src/desktop.cpp src/renderer.cpp src/gpu_context.cpp src/body_statistics_renderer.cpp src/smoke_atlas_renderer.cpp src/smoke_pressure_renderer.cpp src/audio.cpp src/video.cpp src/import_ui.cpp src/app.rc)
target_link_options(VolturaWorlds PRIVATE /MANIFEST:NO)
target_include_directories(VolturaWorlds BEFORE PRIVATE "${DEPS}/agility/build/native/include")
target_link_libraries(VolturaWorlds PRIVATE worlds_core d3d12 dxgi dxguid d3dcompiler dcomp dwmapi comctl32 wtsapi32 avrt winmm mfplat mfreadwrite mfuuid d3d11 shlwapi psapi uxtheme gdiplus)
add_dependencies(VolturaWorlds worlds_shaders)
target_sources(VolturaWorlds PRIVATE src/presets_ui.cpp)
foreach(t worlds_core VolturaWorlds)
  target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8 /EHsc)
endforeach()
file(GLOB_RECURSE WORLD_RUNTIME_INPUTS CONFIGURE_DEPENDS LIST_DIRECTORIES false
  "${CMAKE_SOURCE_DIR}/assets/*" "${CMAKE_SOURCE_DIR}/licenses/*")
set(WORLD_RUNTIME_STAMP "${CMAKE_BINARY_DIR}/runtime-$<CONFIG>.stamp")
add_custom_command(OUTPUT "${WORLD_RUNTIME_STAMP}"
  COMMAND ${CMAKE_COMMAND} -E copy_directory_if_different "${CMAKE_SOURCE_DIR}/licenses" "$<TARGET_FILE_DIR:VolturaWorlds>/licenses"
  COMMAND ${CMAKE_COMMAND} -E copy_directory_if_different "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:VolturaWorlds>/assets"
  COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:VolturaWorlds>/D3D12"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DEPS}/agility/build/native/bin/x64/D3D12Core.dll" "$<TARGET_FILE_DIR:VolturaWorlds>/D3D12/"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DEPS}/agility/build/native/bin/x64/d3d12SDKLayers.dll" "$<TARGET_FILE_DIR:VolturaWorlds>/D3D12/"
  COMMAND ${CMAKE_COMMAND} -E touch "${WORLD_RUNTIME_STAMP}"
  DEPENDS ${WORLD_RUNTIME_INPUTS}
    "${DEPS}/agility/build/native/bin/x64/D3D12Core.dll"
    "${DEPS}/agility/build/native/bin/x64/d3d12SDKLayers.dll"
  VERBATIM)
add_custom_target(worlds_runtime ALL DEPENDS "${WORLD_RUNTIME_STAMP}")
add_dependencies(VolturaWorlds worlds_runtime)

target_link_libraries(VolturaWorlds PRIVATE bcrypt crypt32 wintrust winhttp version)
