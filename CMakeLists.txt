function(shader NAME ENTRY PROFILE)
  set(EXTRA_SHADER_DEPS)
  if(NAME STREQUAL "lattice" OR NAME STREQUAL "mercury")
    list(APPEND EXTRA_SHADER_DEPS "shaders/sculpture_common.hlsli")
  endif()
  if(NAME STREQUAL "ripples")
    list(APPEND EXTRA_SHADER_DEPS "shaders/ripples_surface.hlsli" "shaders/ripples_bottom.hlsli")
  endif()
  if(NAME STREQUAL "bloom_mesh" OR NAME STREQUAL "bloom_fold")
    list(APPEND EXTRA_SHADER_DEPS "shaders/bloom_surface.hlsli")
  endif()
  if(NAME MATCHES "^smoke_filament_")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_filament_shared.hlsli" "shaders/smoke_lighting.hlsli")
  endif()
  if(NAME STREQUAL "fluid_multigrid")
    list(APPEND EXTRA_SHADER_DEPS "shaders/fluid.hlsl" "shaders/smoke_transport.hlsli" "shaders/smoke_lighting.hlsli" "shaders/smoke_sources.hlsli" "shaders/smoke_half.hlsli")
  endif()
  if(NAME STREQUAL "aurora")
    list(APPEND EXTRA_SHADER_DEPS "shaders/aurora_volume.hlsli")
  endif()
  if(NAME STREQUAL "planet")
    list(APPEND EXTRA_SHADER_DEPS "shaders/orbit_rings.hlsli")
  endif()
  if(NAME STREQUAL "planet" OR NAME STREQUAL "orbit")
    list(APPEND EXTRA_SHADER_DEPS "shaders/orbit_terrain.hlsli")
  endif()
  if(NAME STREQUAL "smoke" OR NAME STREQUAL "fluid")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_lighting.hlsli")
  endif()
  if(NAME STREQUAL "smoke_detail" OR NAME STREQUAL "smoke_coordinates")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_detail_clock.hlsli")
  endif()
  if(NAME STREQUAL "fluid" OR NAME STREQUAL "smoke_coordinates")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_transport.hlsli")
  endif()
  if(NAME STREQUAL "smoke_atlas_probe" OR NAME STREQUAL "smoke_atlas_advect")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_atlas.hlsli")
  endif()
  if(NAME STREQUAL "smoke")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_atlas.hlsli")
  endif()
  if(NAME STREQUAL "fluid" OR NAME STREQUAL "smoke_atlas_advect")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_half.hlsli" "shaders/smoke_sources.hlsli")
  endif()
  if(NAME STREQUAL "smoke_atlas_cached" OR NAME STREQUAL "smoke_atlas_cached_curl")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_atlas_advect.hlsl" "shaders/smoke_atlas.hlsli" "shaders/smoke_half.hlsli" "shaders/smoke_sources.hlsli")
  endif()
  if(NAME STREQUAL "smoke_atlas_light")
    list(APPEND EXTRA_SHADER_DEPS "shaders/smoke_atlas.hlsli" "shaders/smoke_lighting.hlsli")
  endif()
  add_custom_command(OUTPUT "${SHADER_DIR}/${NAME}.cso"
    COMMAND "${DXC}" -T ${PROFILE} -E ${ENTRY} -O3 -Ges -Fo "${SHADER_DIR}/${NAME}.cso" "${CMAKE_SOURCE_DIR}/shaders/${NAME}.hlsl"
    DEPENDS "shaders/${NAME}.hlsl" "shaders/common.hlsli" ${EXTRA_SHADER_DEPS} VERBATIM)
  set(SHADER_OUTPUTS ${SHADER_OUTPUTS} "${SHADER_DIR}/${NAME}.cso" PARENT_SCOPE)
endfunction()
